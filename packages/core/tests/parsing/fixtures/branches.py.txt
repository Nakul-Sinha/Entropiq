import sys

DEBUG = "-v" in sys.argv and not sys.flags.quiet
LEVEL = 2 if DEBUG else 1


def classify(value, limit):
    if value is None or value < 0 and limit:
        return "bad"
    elif value == 0:
        return "zero"
    elif value < limit:
        if value % 2:
            return "odd"
        else:
            return "even"
    else:
        return "big"


def scan(rows):
    total = 0
    for row in rows:
        while row:
            with open(row) as handle:
                try:
                    if handle.readable():
                        total += 1
                except (OSError, ValueError):
                    total -= 1
                except Exception:
                    raise
                else:
                    total *= 2
                finally:
                    row = None
    else:
        total = -1
    return total


if __name__ == "__main__":
    for arg in sys.argv[1:]:
        print(classify(len(arg), LEVEL) if arg else scan([arg]))
